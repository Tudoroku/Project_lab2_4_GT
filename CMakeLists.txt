cmake_minimum_required(VERSION 4.1)
project(lab2_4 C)

set(CMAKE_C_STANDARD 23)
option(ENABLE_COVERAGE "Build with code coverage flags" OFF)

if (MSVC)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif ()

set(COVERAGE_RUNTIME_ACTIVE OFF)
if (CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    string(FIND " ${CMAKE_C_FLAGS} ${CMAKE_EXE_LINKER_FLAGS} ${CMAKE_SHARED_LINKER_FLAGS} " "--coverage" COVERAGE_FLAG_POS)
    if (COVERAGE_FLAG_POS GREATER -1)
        set(COVERAGE_RUNTIME_ACTIVE ON)
    endif ()
endif ()

if (ENABLE_COVERAGE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-O0 -g --coverage)
    add_link_options(--coverage)
    set(COVERAGE_RUNTIME_ACTIVE ON)
endif ()

if (COVERAGE_RUNTIME_ACTIVE)
    # Prevent checksum mismatch warnings by removing stale profile files
    # generated by older binaries before rebuilding coverage targets.
    set(COVERAGE_CLEAN_SCRIPT "${CMAKE_BINARY_DIR}/clear_gcda.cmake")
    file(WRITE "${COVERAGE_CLEAN_SCRIPT}" "file(GLOB_RECURSE GCDA_FILES \"${CMAKE_BINARY_DIR}/*.gcda\")\n")
    file(APPEND "${COVERAGE_CLEAN_SCRIPT}" "if(GCDA_FILES)\n")
    file(APPEND "${COVERAGE_CLEAN_SCRIPT}" "  file(REMOVE ${GCDA_FILES})\n")
    file(APPEND "${COVERAGE_CLEAN_SCRIPT}" "endif()\n")

    add_custom_target(coverage_reset_profiles
            COMMAND ${CMAKE_COMMAND} -P "${COVERAGE_CLEAN_SCRIPT}"
            COMMENT "Clearing stale gcov profile data (*.gcda)"
    )
endif ()

set(CORE_SOURCES
        Controller/service.c
        Domain/dispozitiv.c
        Domain/validator.c
        Repository/repo.c
)

set(APP_SOURCES
        ${CORE_SOURCES}
        UI/ui.c
)

set(PROJECT_HEADERS
        Controller/service.h
        Domain/dispozitiv.h
        Domain/validator.h
        Repository/repo.h
        UI/ui.h
        Tests/tests.h
)

add_executable(lab2_4
        main.c
        Tests/tests.c
        ${APP_SOURCES}
        ${PROJECT_HEADERS}
)
target_compile_definitions(lab2_4 PRIVATE LAB2_4_TESTING=1)

add_executable(lab2_4_tests
        Tests/test_main.c
        Tests/tests.c
        ${CORE_SOURCES}
        ${PROJECT_HEADERS}
)
target_compile_definitions(lab2_4_tests PRIVATE LAB2_4_TESTING=1)

if (CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(lab2_4 PRIVATE
            -Wall
            -Wextra
            -Wpedantic
    )

    target_compile_options(lab2_4_tests PRIVATE
            -Wall
            -Wextra
            -Wpedantic
    )
elseif (MSVC)
    target_compile_options(lab2_4 PRIVATE /W4)
    target_compile_options(lab2_4_tests PRIVATE /W4)
endif ()

if (TARGET coverage_reset_profiles)
    add_dependencies(lab2_4 coverage_reset_profiles)
    add_dependencies(lab2_4_tests coverage_reset_profiles)
endif ()

enable_testing()
add_test(NAME lab2_4_unit_tests COMMAND lab2_4_tests)
